-- 경고: NEOVIM 0.10은 최신 안정 릴리스가 아닙니다.
-- 일부 플러그인이 지원을 중단할 가능성이 있습니다.
-- 최신 안정 버전으로 업데이트하는 것을 강력히 권장합니다.

-- ┌────────────────────┐
-- │ MiniMax에 오신 것을 환영합니다 │
-- └────────────────────┘
--
-- 이 설정은 주로 MINI를 사용하도록 설계되었습니다. 별도의 설정 없이도
-- 안정적이고 세련되며 기능이 풍부한 Neovim 경험을 제공합니다. 구조:
--
-- ├ init.lua          시작 시 실행되는 초기 파일(현재 파일)
-- ├ plugin/           시작 시 자동으로 로드되는 파일들
-- ├── 10_options.lua  기본 Neovim 동작
-- ├── 20_keymaps.lua  사용자 정의 매핑
-- ├── 30_mini.lua     MINI 설정
-- ├── 40_plugins.lua  MINI 외의 플러그인
-- ├ snippets/         사용자 정의 스니펫 (데모 파일 포함)
-- ├ after/            플러그인에 의해 추가된 동작을 덮어쓰는 파일들
-- ├── ftplugin/       파일 유형별 동작 (데모 파일 포함)
-- ├── lsp/            언어 서버 설정 (데모 파일 포함)
-- ├── snippets/       높은 우선순위의 스니펫 파일 (데모 파일 포함)
--
-- 설정 파일은 가급적 이 설정이 실행 중인 Neovim 인스턴스에서 루트 디렉토리를 열어
-- 읽는 것을 권장합니다. 이를 통해 설정을 더 잘 이해할 수 있습니다.
-- 이 파일부터 시작하세요. 순서는 상관없지만 위에 나열된 순서를 권장합니다.
-- 설정 탐색 방법:
-- - `<Space>` + `e` + `iokmp` 중 하나 - 'init.lua' 또는 'plugin/' 파일 편집.
-- - 설정 디렉토리 내부: `<Space>ff` (선택기) 또는 `<Space>ed` (탐색기)
-- - `[b`, `]b` 또는 `<Space>fb`로 기존 버퍼 탐색.
--
-- 설정 파일은 사용자 정의가 가능하도록 설계되었습니다. 초기에는 MINI 기반의
-- 작동하는 기준 설정입니다. 이를 수정하여 자신만의 설정을 만드세요. 몇 가지 방법:
-- - 기존 파일을 일관성을 유지하면서 수정합니다.
-- - 일관성을 유지하면서 새로운 파일을 추가합니다.
--   보통 'plugin/' 또는 'after/' 내부에 추가합니다.
--
-- 설정 전반에 걸쳐 이와 같은 문서화 주석을 찾을 수 있습니다.
-- 공통 관례:
--
-- - 사용된 키 표기법은 `:h key-notation`을 참조하세요.
-- - `:h xxx`는 "도움말 태그 xxx의 문서"를 의미합니다. 텍스트를 직접 입력하고
--   Enter를 누르거나 `<Space>fh`를 입력하여 도움말 태그 퍼지 선택기를 엽니다.
-- - "`<Space>fh` 입력"은 "<Space>를 누르고 f를 누른 다음 h를 누름"을 의미합니다.
--   별도의 언급이 없는 한 노멀 모드라고 가정합니다.
-- - "'path/to/file' 참조"는 해당 경로의 파일을 열고 읽는 것을 의미합니다.
-- - `:SomeCommand ...` 또는 `:lua ...`는 해당 명령을 실행하는 것을 의미합니다.

-- 'mini.nvim'을 'mini.deps'에서 관리할 수 있도록 수동으로 부트스트랩합니다.
local mini_path = vim.fn.stdpath('data') .. '/site/pack/deps/start/mini.nvim'
if not vim.loop.fs_stat(mini_path) then
  vim.cmd('echo "Installing `mini.nvim`" | redraw')
  local origin = 'https://github.com/nvim-mini/mini.nvim'
  local clone_cmd = { 'git', 'clone', '--filter=blob:none', origin, mini_path }
  vim.fn.system(clone_cmd)
  vim.cmd('packadd mini.nvim | helptags ALL')
  vim.cmd('echo "Installed `mini.nvim`" | redraw')
end

-- 플러그인 매니저. `now()`/`later()` 헬퍼를 위해 즉시 설정합니다.
-- 사용 예:
-- - `MiniDeps.add('...')` - 설정 내부에서 플러그인 추가
-- - `:DepsUpdate` - 모든 플러그인 업데이트
-- - `:DepsSnapSave` - 현재 활성화된 플러그인의 스냅샷 저장
--
-- 참조:
-- - `:h MiniDeps-overview` - 사용 방법
-- - `:h MiniDeps-commands` - 사용 가능한 모든 명령
-- - 'plugin/30_mini.lua' - 'mini.nvim' 일반에 대한 자세한 내용
require('mini.deps').setup()

-- 스크립트 간에 데이터를 전달할 수 있도록 설정 테이블을 정의합니다.
-- `_G.Config` 및 `Config`로 모두 사용할 수 있는 전역 변수입니다.
_G.Config = {}

-- 사용자 정의 자동 명령 그룹 및 자동 명령을 생성하는 헬퍼를 정의합니다.
-- 자동 명령은 이벤트(버퍼 생성, 옵션 설정 등) 발생 시 실행되는 동작을
-- 정의하는 Neovim의 방식입니다.
--
-- 참조:
-- - `:h autocommand`
-- - `:h nvim_create_augroup()`
-- - `:h nvim_create_autocmd()`
local gr = vim.api.nvim_create_augroup('custom-config', {})
Config.new_autocmd = function(event, pattern, callback, desc)
  local opts = { group = gr, pattern = pattern, callback = callback, desc = desc }
  vim.api.nvim_create_autocmd(event, opts)
end

-- 일부 플러그인 및 'mini.nvim' 모듈은 Neovim이 `nvim -- path/to/file`과 같이
-- 시작된 경우에만 시작 중에 설정이 필요하며, 그렇지 않으면 설정을 지연시켜도 괜찮습니다.
Config.now_if_args = vim.fn.argc(-1) > 0 and MiniDeps.now or MiniDeps.later

-- 일반 ====================================================================

vim.o.signcolumn     = 'yes'      -- 항상 사인 칼럼 표시 (깜빡임 감소)
vim.o.splitbelow     = true       -- 가로 분할은 아래에
vim.o.splitkeep      = 'screen'   -- 창 분할 시 스크롤 감소
vim.o.splitright     = true       -- 세로 분할은 오른쪽에

-- ┌─────────────────┐
-- │ 사용자 정의 매핑 │
-- └─────────────────┘
--
-- 이 파일에는 사용자 정의 일반 및 리더(Leader) 매핑 정의가 포함되어 있습니다.

-- 일반 매핑 ===========================================================

-- 사용자 정의 일반 매핑을 추가하려면 이 섹션을 사용하세요. `:h vim.keymap.set()`을 참조하세요.

-- 노멀 모드 매핑을 생성하는 예시 헬퍼
local nmap = function(lhs, rhs, desc)
  -- `:h vim.keymap.set()` 참조
  vim.keymap.set('n', lhs, rhs, { desc = desc })
end

-- 현재 줄 앞/뒤에 줄 단위로 붙여넣기
-- 사용법: `yiw`로 단어를 복사하고 `]p`로 다음 줄에 붙여넣습니다.
nmap('[p', '<Cmd>exe "put! " . v:register<CR>', '위에 붙여넣기')
nmap(']p', '<Cmd>exe "put "  . v:register<CR>', '아래에 붙여넣기')

-- 많은 일반 매핑이 'mini.basics'에 의해 생성됩니다. 'plugin/30_mini.lua'를 참조하세요.

-- stylua: ignore start
-- 다음 부분(`-- stylua: ignore end`까지)은 더 쉬운 읽기를 위해 수동으로 정렬되었습니다.
-- 이를 유지하거나 `-- stylua` 라인을 제거하여 자동 서식을 사용하세요.

-- 리더(Leader) 매핑 ============================================================

-- Neovim에는 리더 키 개념이 있습니다 (`:h <Leader>` 참조). 주로 "워크플로우"
-- 매핑(텍스트 편집과 반대)에 사용되는 설정 가능한 키입니다.
-- 예: "파일 탐색기 열기", "스크래치 버퍼 생성", "버퍼에서 선택".
--
-- 'plugin/10_options.lua'에서 <Leader>는 <Space>, 즉 리더 키를 누르라는 제안이
-- 있을 때마다 <Space>를 누르세요.
--
-- 이 설정은 "두 개의 키 리더 매핑" 방식을 사용합니다. 첫 번째 키는 의미론적
-- 그룹을 나타내고 두 번째 키는 동작을 실행합니다. 두 키는 보통 기억하기 쉬운
-- 방식으로 선택됩니다.
-- 예: `<Leader>f`는 "찾기(find)" 유형의 동작을 그룹화합니다. `<Leader>ff`는 파일 찾기입니다.
-- 이 섹션을 사용하여 구조적인 방식으로 리더 매핑을 추가하세요.
--
-- 일반적으로 전역 및 로컬 동작이 있는 경우 소문자 두 번째 키는 전역을,
-- 대문자는 로컬을 나타냅니다.
-- 예: `<Leader>fs` / `<Leader>fS` - 작업 공간/문서 LSP 심볼 찾기.
--
-- 많은 매핑이 'plugin/30_mini.lua'에서 설정된 'mini.nvim' 모듈을 사용합니다.

-- Neovim이 `nvim -- path/to/file`과 같이 시작되면 지금 로드하고, 그렇지 않으면 나중에 로드합니다.

-- 자동 완성 및 시그니처 도움말. 비동기 "2단계" 자동 완성을 구현합니다.
-- - 자동 완성 기능을 지원하는 연결된 LSP 서버 기반.
-- - LSP 후보가 없는 경우 폴백 (내장 키워드 자동 완성 기반).
--
-- LSP가 연결된 Insert 모드 사용 예:
-- - LSP가 인식해야 하는 텍스트(예: 변수 이름)를 입력하기 시작합니다.
-- - 100ms 후 후보가 있는 팝업 메뉴가 나타납니다.
-- - `<Tab>` / `<S-Tab>`을 눌러 목록을 아래/위로 탐색합니다. 이는 'mini.keymap'에서 설정됩니다.
--   `<C-n>` / `<C-p>`를 사용할 수도 있습니다.
-- - 탐색 중에 LSP 서버가 후보에 대해 제공할 수 있는 추가 정보를 보여주는 정보 창이 오른쪽에 나타납니다.
--   후보가 100ms 동안 선택된 상태로 유지된 후에 나타납니다. `<C-f>` / `<C-b>`를 사용하여 스크롤하세요.
-- - 항목으로 탐색하면 버퍼 텍스트도 변경됩니다. 만족스러우면 그 뒤에 계속 입력하세요.
--   자동 완성을 완전히 취소하려면 `<C-e>`를 누르세요.
-- - 특수 트리거(보통 `(`)를 누른 후 현재 함수/메서드의 시그니처를 보여주는 창이 나타납니다.
--   입력할 때 업데이트되어 현재 활성화된 매개변수를 보여줍니다.
--
-- 연결된 LSP가 없거나 LSP가 지원하지 않는 위치(예: 주석)의 Insert 모드 사용 예:
-- - 현재 또는 열린 버퍼에 있는 단어를 입력하기 시작합니다.
-- - 100ms 후 후보가 있는 팝업 메뉴가 나타납니다.
-- - `<Tab>` / `<S-Tab>` 또는 `<C-n>` / `<C-p>`로 탐색합니다. 버퍼 텍스트도 업데이트됩니다.
--   선택이 만족스러우면 계속 입력하세요. `<C-e>`로 중단합니다.
--
-- LSP 서버에서 제공하는 스니펫 후보와도 작동합니다. 'mini.snippets'(이 파일에서 설정됨)와
-- 함께 사용할 때 최상의 경험을 제공합니다.
now_if_args(function()
  -- 더 나은 사용자 경험을 위해 LSP 응답의 후처리를 사용자 정의합니다.
  -- 'Text' 제안(보통 시끄러움)을 표시하지 않고 스니펫을 마지막에 표시합니다.
  local process_items_opts = { kind_priority = { Text = -1, Snippet = 99 } }
  local process_items = function(items, base)
    return MiniCompletion.default_process_items(items, base, process_items_opts)
  end
  require('mini.completion').setup({
    lsp_completion = {
      -- 이 설정이 없으면 자동 완성은 `:h 'completefunc'`를 통해 설정됩니다.
      -- 필요하지는 않지만 `:h 'omnifunc'`를 통해 설정하는 것이 더 깔끔하며
      -- (필요할 때만 설정됨) `<C-u>`를 사용할 수 있게 합니다.
      source_func = 'omnifunc',
      auto_setup = false,
      process_items = process_items,
    },
  })

  -- 필요할 때만 LSP 자동 완성을 위해 'omnifunc'를 설정합니다.
  local on_attach = function(ev)
    vim.bo[ev.buf].omnifunc = 'v:lua.MiniCompletion.completefunc_lsp'
  end
  Config.new_autocmd('LspAttach', nil, on_attach, "Set 'omnifunc'")
end)

-- - 언어 파서: 텍스트를 트리로 변환하는 프로그램. 일부는 내장되어 있으며(예: Lua),
--   'nvim-treesitter'는 다른 많은 파서를 제공합니다.
--   참고: 파서를 빌드하고 설치하려면 타사 소프트웨어가 필요합니다.
--   자세한 정보는 MiniMax README의 "Requirements" 섹션 링크를 참조하세요.
-- - 쿼리 파일: 유용한 방식으로 트리에서 정보를 추출하는 방법에 대한 정의
--   (`:h treesitter-query` 참조). 'nvim-treesitter'도 이를 제공하며,
--   'nvim-treesitter-textobjects'는 Neovim 텍스트 오브젝트를 위한 쿼리를
--   제공합니다 (`:h text-objects`, `:h MiniAi.gen_spec.treesitter()` 참조).
--
-- 시작 후 파일('mini.starter' 아님)이 표시되면 지금 이 플러그인들을 추가합니다.
--
-- 문제 해결:
-- - `:checkhealth vim.treesitter nvim-treesitter`를 실행하여 잠재적인 문제를 확인하세요.
-- - Neovim 번들 파서(예: `lua`, `vimdoc`, `markdown` 등)에 대한 쿼리와 관련된 오류가
--   발생하는 경우, `:TSInstall <language>`를 통해 'nvim-treesitter'로 수동으로 설치하세요.
--   필요한 시스템 종속성이 있는지 확인하세요 (MiniMax README의 소프트웨어 요구 사항 섹션 참조).

    source = 'nvim-treesitter/nvim-treesitter',

    hooks = { post_checkout = function() vim.cmd('TSUpdate') end },
    -- Neovim=0.10 지원을 중단하기 직전의 커밋으로 고정
    checkout = '42fc28ba918343ebfd5565147a42a26580579482',
  })
  add({
    source = 'nvim-treesitter/nvim-treesitter-textobjects',
    -- 'nvim-treesitter' 커밋에 해당하는 커밋으로 고정
    checkout = 'b0debd5c424969b4baeabdc8f54db3036c691732',
  })

  -- 파서가 설치되고 자동으로 활성화될 언어 정의
  -- 이를 변경한 후 Neovim을 한 번 재시작하여 필요한 파서를 설치하세요. 추가된
  -- 언어에 대한 파일을 열기 전에 설치가 완료될 때까지 기다리세요.
  local languages = {
    -- 이들은 이미 Neovim과 함께 사전 설치되어 있습니다. 예시로 사용되었습니다.
    'lua',
    'vimdoc',
    'markdown',
    -- 여기에 tree-sitter를 사용하려는 언어를 추가하세요.
    -- 사용 가능한 언어를 보려면:
    -- - `:=require('nvim-treesitter.parsers').available_parsers()` 실행
  }
  require('nvim-treesitter.configs').setup({
    ensure_installed = languages,
    highlight = { enable = true },
    -- 'mini.ai'를 위해 점진적 선택 비활성화
    incremental_selection = { enable = false },
    -- 'mini.ai'를 위해 사용자 정의 텍스트 오브젝트 매핑 비활성화
    textobjects = { enable = false },
    -- 다소 신뢰할 수 없으므로 tree-sitter 기반 들여쓰기 비활성화
    indent = { enable = false },
  })
end)

-- 언어 서버 (Language servers) ===========================================================

-- LSP (Language Server Protocol)는 언어별 도구 제작을 지원하는 일련의 관례입니다.
-- 두 부분이 필요합니다.
-- - 서버 - 언어별 계산을 수행하는 프로그램.
-- - 클라이언트 - 서버에 계산을 요청하고 결과를 보여주는 프로그램.
--
-- 여기서 Neovim 자체가 클라이언트입니다 (`:h vim.lsp` 참조). 언어 서버는
-- OS, CLI 도구 및 선호도에 따라 별도로 설치해야 합니다.
-- 파일 하단의 'mason.nvim'에 대한 참고 사항을 참조하세요.
--
-- Neovim 팀은 대부분의 언어 서버에 대해 일반적으로 사용되는 구성을
-- 'neovim/nvim-lspconfig' 플러그인 내에 수집합니다. 참고: Neovim>=0.11에서
-- `vim.lsp.config` 방식을 사용하도록 대폭 개편되었습니다. 따라서 이 플러그인을
-- 사용하려면 Neovim을 최소 0.11 버전으로 업데이트하는 것을 강력히 권장합니다.
--
-- 시작 후 파일('mini.starter' 아님)이 표시되면 지금 추가합니다.
